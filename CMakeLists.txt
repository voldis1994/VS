cmake_minimum_required(VERSION 3.24)
project(market-reader VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(MR_BUILD_TESTS "Build tests" ON)
option(MR_BUILD_BENCHMARKS "Build benchmarks" ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")

include(GNUInstallDirs)
include(Dependencies)

add_subdirectory(libs/common)

add_subdirectory(libs/position-engine)
add_subdirectory(libs/exit-engine)
add_subdirectory(libs/execution-router)
add_subdirectory(libs/broker-adapters)
add_subdirectory(libs/storage)
add_subdirectory(libs/telemetry)
add_subdirectory(libs/replay)
# C++ sidecar apps archived under legacy-review (not part of Node VS-CORE-01 path)
add_subdirectory(legacy-review/apps/market-core)

if(MR_BUILD_TESTS)
    add_subdirectory(tests)
